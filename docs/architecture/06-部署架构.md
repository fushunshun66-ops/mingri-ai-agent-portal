### 3.2 docker-compose.yml

```yaml

services:
  # ── 反向代理 ──
  nginx:
    image: nginx:1.25-alpine
    container_name: portal-nginx

      - "${NGINX_PORT:-80}:80"
      - "${NGINX_SSL_PORT:-443}:443"

      - ./docker/nginx/nginx.conf:/etc/nginx/nginx.conf:ro
```

### 3.3 后端 Dockerfile

```dockerfile
# backend/Dockerfile
FROM python:3.12-slim

ARG SERVICE
ENV SERVICE_NAME=${SERVICE}

WORKDIR /app

# 安装依赖
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# 复制公共库和指定服务代码
COPY common/ ./common/
COPY ${SERVICE}/ ./${SERVICE}/

# 暴露端口由环境变量决定
CMD ["sh", "-c", "uvicorn ${SERVICE_NAME}.main:app --host 0.0.0.0 --port ${SERVICE_PORT:-8000} --reload"]
```


### 3.5 Makefile

```makefile
.PHONY: up down build logs restart clean

# 启动所有服务
up:
	docker compose up -d

# 启动 + 监控服务
up-full:
	docker compose --profile monitoring up -d

# 停止所有服务
down:
	docker compose down

# 重新构建并启动
build:
	docker compose build --no-cache
	docker compose up -d

# 查看日志
logs:
	docker compose logs -f

# 查看指定服务日志
logs-%:
	docker compose logs -f $*

# 重启指定服务
restart-%:
	docker compose restart $*

# 进入指定服务容器
exec-%:
	docker compose exec $* sh

# 数据库迁移
migrate:
	docker compose exec users alembic upgrade head
	docker compose exec agents alembic upgrade head
	docker compose exec adapters alembic upgrade head
	docker compose exec chat alembic upgrade head
	docker compose exec monitor alembic upgrade head

# 清理所有数据和容器
clean:
	docker compose down -v
	rm -rf docker/postgres/data/*
```

### 5.2 GitHub Actions 配置

```yaml
# .github/workflows/ci.yml
name: CI

on:
  push:
    branches: [main, dev]
  pull_request:
    branches: [main]

jobs:
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install Python deps
        run: pip install ruff

      - name: Lint Python
        run: ruff check backend/

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: "20"

      - name: Install Node deps
        working-directory: frontend
        run: npm ci

      - name: Lint Frontend
        working-directory: frontend
        run: npm run lint

  test-backend:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: pgvector/pgvector:pg14
        env:
          POSTGRES_DB: test_db
          POSTGRES_USER: test
          POSTGRES_PASSWORD: test
        ports:
          - 5432:5432
      redis:
        image: redis:7-alpine
        ports:
          - 6379:6379

    steps:
      - uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install deps
        run: pip install -r backend/requirements-dev.txt

      - name: Run tests with coverage
        env:
          DATABASE_URL: postgresql+asyncpg://test:test@localhost:5432/test_db
          REDIS_URL: redis://localhost:6379/0
        run: |
          cd backend
          pytest --cov=. --cov-report=xml --cov-report=term -n auto
          coverage report --fail-under=80

      - name: Upload coverage
        uses: codecov/codecov-action@v3
        with:
          file: backend/coverage.xml

  test-frontend:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: "20"

      - name: Install deps
        working-directory: frontend
        run: npm ci

      - name: Run tests
        working-directory: frontend
        run: npm run test -- --coverage

  build:
    needs: [lint, test-backend, test-frontend]
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Build and push Docker images
        run: |
          docker build -t portal-users:${{ github.sha }} ./backend --build-arg SERVICE=users
          docker build -t portal-agents:${{ github.sha }} ./backend --build-arg SERVICE=agents
          docker build -t portal-adapters:${{ github.sha }} ./backend --build-arg SERVICE=adapters
          docker build -t portal-chat:${{ github.sha }} ./backend --build-arg SERVICE=chat
          docker build -t portal-monitor:${{ github.sha }} ./backend --build-arg SERVICE=monitor
          # Push to registry (Docker Hub / GHCR / ACR / TCR)
```

### 5.3 部署策略

| 环境 | 触发方式 | 部署方式 |
|---|---|---|
| **Development** | push to `dev` 分支 | Docker Compose 自动更新 |
| **Staging** | push to `main` 分支 | K8s / Compose 自动部署 |
| **Production** | 手动审批 + Release Tag | K8s 滚动更新 / 蓝绿部署 |

### 5.4 数据库迁移策略

- 使用 **Alembic** 管理数据库迁移
- 迁移脚本随代码一起提交，CI 中自动执行
- **回滚策略**：每次迁移包含 `upgrade()` 和 `downgrade()`
- **零停机迁移**（生产）：
  - 先加列（nullable），不改已有列
  - 代码同时兼容新旧 Schema
  - 数据回填后，下次发布移除旧代码

## 6. 健康检查与监控

### 6.1 健康检查端点

每个服务暴露以下端点：

```
GET /health       — 存活探针（liveness）：进程是否运行
GET /ready        — 就绪探针（readiness）：DB/Redis/依赖是否就绪
GET /metrics      — Prometheus 指标
```

### 6.2 关键告警规则

| 告警 | 条件 | 级别 |
|---|---|---|
| 服务不可用 | `up == 0` 持续 1 分钟 | Critical |
| API 错误率过高 | 5xx 错误率 > 5% 持续 5 分钟 | Critical |
| API 延迟过高 | P95 > 2s 持续 5 分钟 | Warning |
| 数据库连接池耗尽 | 活跃连接 > 80% 池大小 | Warning |
| Redis 内存使用 > 80% | `redis_memory_used / redis_memory_max > 0.8` | Warning |
| 磁盘使用 > 85% | `disk_used_percent > 85` | Warning |

### 6.3 Grafana 看板建议

| 看板 | 内容 |
|---|---|
| **服务概览** | 各服务实例数量、CPU/内存、请求QPS、错误率 |
| **API 详情** | 各端点延迟分布（P50/P95/P99）、请求量排行 |
| **对话统计** | 活跃会话数、消息量、流式延迟、Token 消耗 |
| **平台连接** | 各平台连接状态、调用成功率、响应时间 |
| **业务指标** | 新注册用户、Agent 安装数、活跃用户、市场搜索量 |
